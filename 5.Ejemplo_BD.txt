-- ✅ CREAR BASE DE DATOS Y USARLA
CREATE DATABASE tienda_robotica;
USE tienda_robotica;

-- ✅ TABLA DE PRODUCTOS
CREATE TABLE productos (
    id_producto INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT,
    precio DECIMAL(10,2),
    stock INT,
    categoria VARCHAR(50)
);

-- ✅ TABLA DE CLIENTES
CREATE TABLE clientes (
    id_cliente INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100),
    correo VARCHAR(100),
    telefono VARCHAR(20),
    direccion TEXT
);

-- ✅ TABLA DE VENTAS (CON CLAVES FORÁNEAS)
CREATE TABLE ventas (
    id_venta INT AUTO_INCREMENT PRIMARY KEY,
    id_cliente INT,
    id_producto INT,
    cantidad INT,
    fecha_venta DATE,
    FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente),
    FOREIGN KEY (id_producto) REFERENCES productos(id_producto)
);

-- --------------------------------------------------
-- 🔎 10 SELECTS
-- --------------------------------------------------

-- 1. Ver todos los productos
SELECT * FROM productos;

-- 2. Ver productos con stock menor a 10
SELECT * FROM productos WHERE stock < 10;

-- 3. Ver clientes ordenados alfabéticamente
SELECT * FROM clientes ORDER BY nombre;

-- 4. Ver ventas con JOIN
SELECT v.id_venta, c.nombre, p.nombre, v.cantidad, v.fecha_venta
FROM ventas v
JOIN clientes c ON v.id_cliente = c.id_cliente
JOIN productos p ON v.id_producto = p.id_producto;

-- 5. Buscar productos de la categoría "Sensores"
SELECT * FROM productos WHERE categoria = 'Sensores';

-- 6. Buscar cliente por correo
SELECT * FROM clientes WHERE correo = 'marta@example.com';

-- 7. Ver total de productos vendidos por nombre
SELECT p.nombre, SUM(v.cantidad) AS total_vendido
FROM ventas v
JOIN productos p ON v.id_producto = p.id_producto
GROUP BY p.nombre;

-- 8. Ver ventas de un cliente específico
SELECT * FROM ventas WHERE id_cliente = 3;

-- 9. Contar cantidad de clientes
SELECT COUNT(*) AS total_clientes FROM clientes;

-- 10. Ver los productos con precio entre 10000 y 30000
SELECT * FROM productos WHERE precio BETWEEN 10000 AND 30000;
