-- =====================================================
-- TABLA 1: CLIENTES
-- Almacena la informacion de los clientes
-- =====================================================

CREATE TABLE CLIENTES (

    -- Identificador unico del cliente
    ID_CLIENTE NUMBER PRIMARY KEY,

    -- Nombre del cliente
    NOMBRE VARCHAR2(50) NOT NULL,

    -- Apellido del cliente
    APELLIDO VARCHAR2(50) NOT NULL,

    -- Numero telefonico
    TELEFONO VARCHAR2(20),

    -- Correo electronico
    CORREO VARCHAR2(100)

);

-- =====================================================
-- TABLA 2: BICICLETAS
-- Almacena las bicicletas asociadas a cada cliente
-- =====================================================

CREATE TABLE BICICLETAS (

    -- Identificador unico de la bicicleta
    ID_BICICLETA NUMBER PRIMARY KEY,

    -- Marca de la bicicleta
    MARCA VARCHAR2(50) NOT NULL,

    -- Modelo de la bicicleta
    MODELO VARCHAR2(50) NOT NULL,

    -- Precio de la bicicleta
    PRECIO NUMBER(10,2) NOT NULL,

    -- Identificador del cliente propietario
    ID_CLIENTE NUMBER,

    -- Llave foranea que relaciona las dos tablas
    CONSTRAINT FK_BICICLETAS_CLIENTES
    FOREIGN KEY (ID_CLIENTE)
    REFERENCES CLIENTES(ID_CLIENTE)

);

-- =====================================================
-- 7. CONSULTAR CLIENTES Y SUS BICICLETAS
-- Utilizamos INNER JOIN para relacionar las tablas
-- =====================================================

SELECT
    CLIENTES.ID_CLIENTE,
    CLIENTES.NOMBRE,
    CLIENTES.APELLIDO,
    BICICLETAS.MARCA,
    BICICLETAS.MODELO,
    BICICLETAS.PRECIO

FROM CLIENTES

INNER JOIN BICICLETAS
ON CLIENTES.ID_CLIENTE = BICICLETAS.ID_CLIENTE;

* Bicicletas por cliente

SELECT    C.NOMBRE,
    C.APELLIDO,
    COUNT(B.ID_BICICLETA) AS TOTAL_BICICLETAS
FROM CLIENTES C
LEFT JOIN BICICLETAS B
ON C.ID_CLIENTE = B.ID_CLIENTE
GROUP BY C.NOMBRE, C.APELLIDO;
