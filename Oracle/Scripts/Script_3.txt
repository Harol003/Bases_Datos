sudo su - oracle

sqlplus / as sysdba



-- ==========================================================
-- PROYECTO: SISTEMA DE GESTION DE UN PARQUEADERO
-- PRIMERA PARTE: CREACION DEL USUARIO
-- ==========================================================

-- Seleccionar la base de datos conectable
ALTER SESSION SET CONTAINER = FREEPDB1;

-- Crear el usuario PARQUEADERO con su contraseña
CREATE USER PARQUEADERO IDENTIFIED BY Parque123;

-- Permitir que el usuario pueda conectarse a Oracle
GRANT CREATE SESSION TO PARQUEADERO;

-- Permitir que pueda crear tablas
GRANT CREATE TABLE TO PARQUEADERO;

-- Permitir que pueda crear vistas
GRANT CREATE VIEW TO PARQUEADERO;

-- Permitir utilizar espacio de almacenamiento
GRANT UNLIMITED TABLESPACE TO PARQUEADERO;

-- Comprobar que el usuario fue creado
SELECT USERNAME
FROM DBA_USERS
WHERE USERNAME = 'PARQUEADERO';

-- Salir del administrador
EXIT;


# Conectarse a Oracle utilizando el usuario PARQUEADERO
sqlplus PARQUEADERO/Parque123@localhost:1521/FREEPDB1

SELECT USER FROM DUAL;

TABLAS 
**************************************************************


-- ==========================================================
-- SEGUNDA PARTE: CREACION DE LAS TABLAS
-- ==========================================================

-- Crear la tabla VEHICULOS
CREATE TABLE VEHICULOS (

    -- Identificador unico del vehiculo
    ID_VEHICULO NUMBER PRIMARY KEY,

    -- Placa del vehiculo
    PLACA VARCHAR2(10) NOT NULL UNIQUE,

    -- Marca del vehiculo
    MARCA VARCHAR2(30),

    -- Modelo o año del vehiculo
    MODELO VARCHAR2(30),

    -- Color del vehiculo
    COLOR VARCHAR2(20)
);

-- Crear la tabla INGRESOS
CREATE TABLE INGRESOS (

    -- Identificador unico del ingreso
    ID_INGRESO NUMBER PRIMARY KEY,

    -- Identificador del vehiculo relacionado
    ID_VEHICULO NUMBER NOT NULL,

    -- Fecha y hora en que ingreso el vehiculo
    FECHA_INGRESO TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,

    -- Fecha y hora en que salio el vehiculo
    FECHA_SALIDA TIMESTAMP,

    -- Valor pagado por el servicio
    VALOR_PAGADO NUMBER(10,2),

    -- Crear la clave foranea
    -- Relaciona INGRESOS con VEHICULOS
    CONSTRAINT FK_INGRESOS_VEHICULOS
    FOREIGN KEY (ID_VEHICULO)
    REFERENCES VEHICULOS(ID_VEHICULO)
);

-- Comprobar la creacion de las tablas
SELECT TABLE_NAME
FROM USER_TABLES;

***************************************************************
CONSULTAS AVANZADAS
***************************************************************


-- Consultar cuantos ingresos tiene cada vehiculo
SELECT
    V.PLACA,
    V.MARCA,
    COUNT(I.ID_INGRESO) AS TOTAL_INGRESOS
FROM VEHICULOS V
LEFT JOIN INGRESOS I
ON V.ID_VEHICULO = I.ID_VEHICULO
GROUP BY V.PLACA, V.MARCA;

-- Calcular el total de dinero recaudado
SELECT SUM(VALOR_PAGADO) AS TOTAL_RECAUDADO
FROM INGRESOS;

-- Consultar los vehiculos que todavia estan en el parqueadero
-- FECHA_SALIDA IS NULL significa que no se ha registrado su salida
SELECT
    V.PLACA,
    V.MARCA,
    I.FECHA_INGRESO
FROM VEHICULOS V
INNER JOIN INGRESOS I
ON V.ID_VEHICULO = I.ID_VEHICULO
WHERE I.FECHA_SALIDA IS NULL;

-- Consultar todos los movimientos ordenados por fecha
SELECT *
FROM INGRESOS
ORDER BY FECHA_INGRESO DESC;
