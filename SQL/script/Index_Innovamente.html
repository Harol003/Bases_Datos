Reto 3 - INNER JOIN

Obtén el nombre del estudiante y el nombre de su grado.

SELECT E.Nombre, E.Apellido, G.NombreGrado FROM Estudiantes E INNER JOIN Grados G ON E.IdGrado = G.IdGrado;
Reto 4 - LEFT JOIN

Encuentra los estudiantes que todavía no tienen calificaciones.

SELECT E.Nombre, E.Apellido FROM Estudiantes E LEFT JOIN Calificaciones C ON E.IdEstudiante = C.IdEstudiante WHERE C.IdCalificacion IS NULL;
Reto 5 - GROUP BY

Calcula cuántos estudiantes hay en cada grado.

SELECT G.NombreGrado, COUNT(E.IdEstudiante) AS Cantidad FROM Grados G LEFT JOIN Estudiantes E ON G.IdGrado = E.IdGrado GROUP BY G.NombreGrado;

Obtén el promedio de cada estudiante.

SELECT E.Nombre, E.Apellido, AVG(C.Nota) AS Promedio FROM Estudiantes E INNER JOIN Calificaciones C ON E.IdEstudiante = C.IdEstudiante GROUP BY E.Nombre, E.Apellido;
