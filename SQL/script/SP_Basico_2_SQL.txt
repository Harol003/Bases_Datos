/*
=========================================================
TABLA CATEGORIAS
=========================================================
*/

CREATE TABLE Categorias
(
    id_categoria INT IDENTITY(1,1) PRIMARY KEY,

    nombre_categoria VARCHAR(100)
    NOT NULL
    UNIQUE
);
GO

/*
=========================================================
TABLA PRODUCTOS
=========================================================
*/

CREATE TABLE Productos
(
    id_producto INT IDENTITY(1,1) PRIMARY KEY,

    nombre_producto VARCHAR(100)
    NOT NULL,

    talla VARCHAR(10)
    NOT NULL,

    color VARCHAR(50)
    NOT NULL,

    precio DECIMAL(10,2)
    CHECK(precio > 0),

    stock INT
    DEFAULT 0
    CHECK(stock >= 0),

    id_categoria INT,

    CONSTRAINT FK_PRODUCTOS_CATEGORIAS
    FOREIGN KEY(id_categoria)
    REFERENCES Categorias(id_categoria)
);
GO

/*
=========================================================
TABLA CLIENTES
=========================================================
*/

CREATE TABLE Clientes
(
    id_cliente INT IDENTITY(1,1) PRIMARY KEY,

    nombre VARCHAR(100)
    NOT NULL,

    telefono VARCHAR(20),

    correo VARCHAR(100)
    UNIQUE,

    ciudad VARCHAR(100)
);
GO

/*
=========================================================
TABLA EMPLEADOS
=========================================================
*/

CREATE TABLE Empleados
(
    id_empleado INT IDENTITY(1,1) PRIMARY KEY,

    nombre VARCHAR(100)
    NOT NULL,

    cargo VARCHAR(50),

    salario DECIMAL(10,2)
    CHECK(salario > 0)
);
GO

/*
=========================================================
TABLA VENTAS
=========================================================
*/

CREATE TABLE Ventas
(
    id_venta INT IDENTITY(1,1) PRIMARY KEY,

    fecha_venta DATE
    DEFAULT GETDATE(),

    total DECIMAL(10,2),

    id_cliente INT,

    id_empleado INT,

    CONSTRAINT FK_VENTAS_CLIENTES
    FOREIGN KEY(id_cliente)
    REFERENCES Clientes(id_cliente),

    CONSTRAINT FK_VENTAS_EMPLEADOS
    FOREIGN KEY(id_empleado)
    REFERENCES Empleados(id_empleado)
);
GO

/*
=========================================================
TABLA DETALLEVENTAS

Permite que una venta tenga varios productos.
=========================================================
*/

CREATE TABLE DetalleVentas
(
    id_detalle INT IDENTITY(1,1) PRIMARY KEY,

    cantidad INT
    CHECK(cantidad > 0),

    subtotal DECIMAL(10,2),

    id_venta INT,

    id_producto INT,

    CONSTRAINT FK_DETALLE_VENTA
    FOREIGN KEY(id_venta)
    REFERENCES Ventas(id_venta),

    CONSTRAINT FK_DETALLE_PRODUCTO
    FOREIGN KEY(id_producto)
    REFERENCES Productos(id_producto)
);
GO

/*
=========================================================
CONSULTA INNER JOIN
=========================================================
*/

SELECT
P.nombre_producto,
C.nombre_categoria,
P.precio
FROM Productos P
INNER JOIN Categorias C
ON P.id_categoria = C.id_categoria;
GO

/*
=========================================================
CONSULTA DE VENTAS COMPLETAS
=========================================================
*/

SELECT
V.id_venta,
CL.nombre AS cliente,
E.nombre AS empleado,
V.total
FROM Ventas V
INNER JOIN Clientes CL
ON V.id_cliente = CL.id_cliente
INNER JOIN Empleados E
ON V.id_empleado = E.id_empleado;
GO

/*
=========================================================
LEFT JOIN
=========================================================
*/

SELECT
CL.nombre,
V.id_venta
FROM Clientes CL
LEFT JOIN Ventas V
ON CL.id_cliente = V.id_cliente;
GO

/*
=========================================================
RIGHT JOIN
=========================================================
*/

SELECT
CL.nombre,
V.id_venta
FROM Clientes CL
RIGHT JOIN Ventas V
ON CL.id_cliente = V.id_cliente;
GO

/*
=========================================================
GROUP BY

TOTAL VENDIDO POR EMPLEADO
=========================================================
*/

SELECT
E.nombre,
SUM(V.total) AS total_vendido
FROM Ventas V
INNER JOIN Empleados E
ON V.id_empleado = E.id_empleado
GROUP BY E.nombre;
GO

/*
=========================================================
SP MOSTRAR PRODUCTOS
=========================================================
*/

CREATE PROCEDURE sp_mostrar_productos
AS
BEGIN

    SELECT *
    FROM Productos;

END;
GO

/*
=========================================================
SP BUSCAR PRODUCTO POR CATEGORIA
=========================================================
*/

CREATE PROCEDURE sp_buscar_categoria
    @categoria INT
AS
BEGIN

    SELECT *
    FROM Productos
    WHERE id_categoria = @categoria;

END;
GO

/*
=========================================================
SP INSERTAR PRODUCTO
=========================================================
*/

CREATE PROCEDURE sp_insertar_producto

    @nombre VARCHAR(100),
    @talla VARCHAR(10),
    @color VARCHAR(50),
    @precio DECIMAL(10,2),
    @stock INT,
    @categoria INT

AS
BEGIN

    INSERT INTO Productos
    (
        nombre_producto,
        talla,
        color,
        precio,
        stock,
        id_categoria
    )
    VALUES
    (
        @nombre,
        @talla,
        @color,
        @precio,
        @stock,
        @categoria
    );

END;
GO

/*
=========================================================
SP ACTUALIZAR PRECIO
=========================================================
*/

CREATE PROCEDURE sp_actualizar_precio

    @id INT,
    @precio DECIMAL(10,2)

AS
BEGIN

    UPDATE Productos

    SET precio = @precio

    WHERE id_producto = @id;

END;
GO

/*
=========================================================
SP ELIMINAR PRODUCTO
=========================================================
*/

CREATE PROCEDURE sp_eliminar_producto

    @id INT

AS
BEGIN

    DELETE FROM Productos

    WHERE id_producto = @id;

END;
GO

/*
=========================================================
REPORTE DE VENTAS
=========================================================
*/

CREATE PROCEDURE sp_reporte_ventas
AS
BEGIN

    SELECT

        V.id_venta,
        CL.nombre AS cliente,
        E.nombre AS empleado,
        V.total

    FROM Ventas V

    INNER JOIN Clientes CL
    ON V.id_cliente = CL.id_cliente

    INNER JOIN Empleados E
    ON V.id_empleado = E.id_empleado;

END;
GO

EXEC sp_reporte_ventas;
