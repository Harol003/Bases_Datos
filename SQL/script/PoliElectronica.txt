/* =========================================================
   TABLA 1 - CLIENTES

   ESTA TABLA ALMACENA LOS DATOS
   DE LOS CLIENTES DE LA TIENDA.
========================================================= */

CREATE TABLE Clientes(

    /* LLAVE PRIMARIA */
    id_cliente INT PRIMARY KEY IDENTITY(1,1),

    /* NOMBRE DEL CLIENTE */
    nombre VARCHAR(100) NOT NULL,

    /* APELLIDO DEL CLIENTE */
    apellido VARCHAR(100) NOT NULL,

    /* TELEFONO UNICO */
    telefono VARCHAR(20) UNIQUE,

    /* CORREO ELECTRONICO */
    correo VARCHAR(100),

    /* CIUDAD */
    ciudad VARCHAR(50)
);
GO



/* =========================================================
   TABLA 2 - EMPLEADOS

   GUARDA LA INFORMACION
   DE LOS EMPLEADOS.
========================================================= */

CREATE TABLE Empleados(

    id_empleado INT PRIMARY KEY IDENTITY(1,1),

    nombre VARCHAR(100) NOT NULL,

    cargo VARCHAR(50),

    salario DECIMAL(10,2),

    telefono VARCHAR(20)
);
GO



/* =========================================================
   TABLA 3 - CATEGORIAS

   CLASIFICA LOS PRODUCTOS.
========================================================= */

CREATE TABLE Categorias(

    id_categoria INT PRIMARY KEY IDENTITY(1,1),

    nombre_categoria VARCHAR(100) NOT NULL
);
GO



/* =========================================================
   TABLA 4 - PROVEEDORES

   ALMACENA LOS PROVEEDORES
   DE PRODUCTOS ELECTRONICOS.
========================================================= */

CREATE TABLE Proveedores(

    id_proveedor INT PRIMARY KEY IDENTITY(1,1),

    nombre_proveedor VARCHAR(100) NOT NULL,

    telefono VARCHAR(20),

    ciudad VARCHAR(50)
);
GO



/* =========================================================
   TABLA 5 - PRODUCTOS

   CONTIENE TODOS LOS PRODUCTOS
   DE LA TIENDA.
========================================================= */

CREATE TABLE Productos(

    id_producto INT PRIMARY KEY IDENTITY(1,1),

    nombre_producto VARCHAR(100) NOT NULL,

    precio DECIMAL(10,2) NOT NULL,

    stock INT NOT NULL,

    id_categoria INT,

    id_proveedor INT,

    /* RESTRICCION:
       EL STOCK NO PUEDE SER NEGATIVO */
    CHECK(stock >= 0),

    /* RESTRICCION:
       EL PRECIO DEBE SER MAYOR A CERO */
    CHECK(precio > 0),

    /* LLAVE FORANEA HACIA CATEGORIAS */
    FOREIGN KEY(id_categoria)
    REFERENCES Categorias(id_categoria),

    /* LLAVE FORANEA HACIA PROVEEDORES */
    FOREIGN KEY(id_proveedor)
    REFERENCES Proveedores(id_proveedor)
);
GO



/* =========================================================
   TABLA 6 - VENTAS

   GUARDA LAS VENTAS
   REALIZADAS EN LA TIENDA.
========================================================= */

CREATE TABLE Ventas(

    id_venta INT PRIMARY KEY IDENTITY(1,1),

    id_cliente INT,

    id_empleado INT,

    fecha DATE NOT NULL,

    total DECIMAL(10,2),

    /* LLAVE FORANEA CLIENTE */
    FOREIGN KEY(id_cliente)
    REFERENCES Clientes(id_cliente),

    /* LLAVE FORANEA EMPLEADO */
    FOREIGN KEY(id_empleado)
    REFERENCES Empleados(id_empleado)
);
GO



/* =========================================================
   TABLA 7 - DETALLEVENTA

   ESTA TABLA RELACIONA
   PRODUCTOS CON VENTAS.
========================================================= */

CREATE TABLE DetalleVenta(

    id_detalle INT PRIMARY KEY IDENTITY(1,1),

    id_venta INT,

    id_producto INT,

    cantidad INT,

    subtotal DECIMAL(10,2),

    FOREIGN KEY(id_venta)
    REFERENCES Ventas(id_venta),

    FOREIGN KEY(id_producto)
    REFERENCES Productos(id_producto)
);
GO



/* =========================================================
   TABLA 8 - SERVICIOS TECNICOS

   GUARDA LAS REPARACIONES
   Y SERVICIOS TECNICOS.
========================================================= */

CREATE TABLE ServiciosTecnicos(

    id_servicio INT PRIMARY KEY IDENTITY(1,1),

    id_cliente INT,

    descripcion VARCHAR(200),

    estado VARCHAR(50),

    fecha_ingreso DATE,

    FOREIGN KEY(id_cliente)
    REFERENCES Clientes(id_cliente)
);
GO

/* =========================================================

SELECT

Productos.nombre_producto,

SUM(DetalleVenta.subtotal) AS Total

FROM Productos

INNER JOIN DetalleVenta
ON Productos.id_producto = DetalleVenta.id_producto

GROUP BY Productos.nombre_producto

HAVING SUM(DetalleVenta.subtotal) > 50000;
GO

CREATE VIEW vw_ventas_generales
AS

Clientes.nombre AS Cliente,

Productos.nombre_producto AS Producto,

FROM Ventas

INNER JOIN Clientes
ON Ventas.id_cliente = Clientes.id_cliente

INNER JOIN DetalleVenta
ON Ventas.id_venta = DetalleVenta.id_venta

INNER JOIN Productos
ON DetalleVenta.id_producto = Productos.id_producto;
GO

/* =========================================================

   INSERTAR VENTAS
========================================================= */

CREATE PROCEDURE sp_insertar_venta

@id_cliente INT,
@id_empleado INT,
@fecha DATE,
@total DECIMAL(10,2)

AS

BEGIN

INSERT INTO Ventas
(id_cliente,id_empleado,fecha,total)

VALUES
(@id_cliente,@id_empleado,@fecha,@total);

END;
GO

/* =========================================================
   TRIGGER

   DESCUENTA STOCK AUTOMATICAMENTE
========================================================= */

CREATE TRIGGER trg_actualizar_stock

ON DetalleVenta

AFTER INSERT

AS

BEGIN

UPDATE Productos

SET stock = stock - inserted.cantidad

FROM Productos

INNER JOIN inserted
ON Productos.id_producto = inserted.id_producto;

END;
GO
