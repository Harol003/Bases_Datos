/*========================================================
CREACION DE LA BASE DE DATOS
========================================================*/

CREATE DATABASE BikeZoneDB;
GO

USE BikeZoneDB;
GO

/*========================================================
TABLA CLIENTES
========================================================*/

CREATE TABLE Clientes(
    id_cliente INT PRIMARY KEY IDENTITY(1,1),
    nombre VARCHAR(100) NOT NULL,
    telefono VARCHAR(20),
    correo VARCHAR(100),
    direccion VARCHAR(200)
);

GO

/*========================================================
TABLA EMPLEADOS
========================================================*/

CREATE TABLE Empleados(
    id_empleado INT PRIMARY KEY IDENTITY(1,1),
    nombre VARCHAR(100) NOT NULL,
    cargo VARCHAR(50),
    telefono VARCHAR(20)
);

GO

/*========================================================
TABLA CATEGORIAS
========================================================*/

CREATE TABLE Categorias(
    id_categoria INT PRIMARY KEY IDENTITY(1,1),
    nombre_categoria VARCHAR(100) NOT NULL
);

GO

/*========================================================
TABLA PRODUCTOS
========================================================*/

CREATE TABLE Productos(
    id_producto INT PRIMARY KEY IDENTITY(1,1),
    nombre_producto VARCHAR(100) NOT NULL,
    precio DECIMAL(10,2),
    stock INT,
    id_categoria INT,

    CONSTRAINT FK_Productos_Categorias
    FOREIGN KEY(id_categoria)
    REFERENCES Categorias(id_categoria)
);

GO

/*========================================================
TABLA VENTAS
========================================================*/

CREATE TABLE Ventas(
    id_venta INT PRIMARY KEY IDENTITY(1,1),
    fecha DATE,
    total DECIMAL(10,2),
    id_cliente INT,
    id_empleado INT,

    CONSTRAINT FK_Ventas_Clientes
    FOREIGN KEY(id_cliente)
    REFERENCES Clientes(id_cliente),

    CONSTRAINT FK_Ventas_Empleados
    FOREIGN KEY(id_empleado)
    REFERENCES Empleados(id_empleado)
);

GO

/*========================================================
TABLA DETALLE VENTA
========================================================*/

CREATE TABLE DetalleVenta(
    id_detalle INT PRIMARY KEY IDENTITY(1,1),
    cantidad INT,
    subtotal DECIMAL(10,2),
    id_venta INT,
    id_producto INT,

    CONSTRAINT FK_DetalleVenta_Ventas
    FOREIGN KEY(id_venta)
    REFERENCES Ventas(id_venta),

    CONSTRAINT FK_DetalleVenta_Productos
    FOREIGN KEY(id_producto)
    REFERENCES Productos(id_producto)
);

GO

/*========================================================
TABLA SERVICIOS TECNICOS
========================================================*/

CREATE TABLE ServiciosTecnicos(
    id_servicio INT PRIMARY KEY IDENTITY(1,1),
    descripcion VARCHAR(200),
    estado VARCHAR(50),
    fecha_ingreso DATE,
    fecha_salida DATE,
    id_cliente INT,

    CONSTRAINT FK_Servicios_Clientes
    FOREIGN KEY(id_cliente)
    REFERENCES Clientes(id_cliente)
);

GO

/*========================================================
CONSULTA 2
PRODUCTOS Y CATEGORIAS
========================================================*/

SELECT
P.nombre_producto,
P.precio,
P.stock,
C.nombre_categoria
FROM Productos P
INNER JOIN Categorias C
ON P.id_categoria = C.id_categoria;

GO

/*========================================================
CONSULTA 3
VENTAS COMPLETAS
========================================================*/

SELECT
V.id_venta,
V.fecha,
V.total,
C.nombre AS cliente,
E.nombre AS empleado
FROM Ventas V
INNER JOIN Clientes C
ON V.id_cliente = C.id_cliente
INNER JOIN Empleados E
ON V.id_empleado = E.id_empleado;

GO

/*========================================================
CONSULTA 4
TOTAL VENDIDO POR PRODUCTO
========================================================*/

SELECT
P.nombre_producto,
SUM(DV.subtotal) AS total_vendido
FROM DetalleVenta DV
INNER JOIN Productos P
ON DV.id_producto = P.id_producto
GROUP BY P.nombre_producto;

GO

/*========================================================
CONSULTA 5
CLIENTES CON SERVICIO TECNICO
========================================================*/

SELECT
C.nombre,
S.descripcion,
S.estado
FROM ServiciosTecnicos S
INNER JOIN Clientes C
ON S.id_cliente = C.id_cliente;

GO

/*========================================================
CONSULTA 6
PRODUCTOS CON STOCK BAJO
========================================================*/

SELECT *
FROM Productos
WHERE stock < 5;

GO

/*========================================================
CONSULTA 7
TOTAL VENDIDO POR EMPLEADO
========================================================*/

SELECT
E.nombre,
SUM(V.total) AS total_vendido
FROM Ventas V
INNER JOIN Empleados E
ON V.id_empleado = E.id_empleado
GROUP BY E.nombre;

GO

/*========================================================
CONSULTA 8
SERVICIOS PENDIENTES
========================================================*/

SELECT *
FROM ServiciosTecnicos
WHERE estado='Pendiente';

GO
