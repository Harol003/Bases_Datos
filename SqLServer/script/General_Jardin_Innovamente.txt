/* ================================================================
   2. CREAR TABLA: PADRES DE FAMILIA
================================================================ */

CREATE TABLE PadresFamilia
(
    IdPadre INT IDENTITY(1,1) PRIMARY KEY,

    Nombre VARCHAR(50) NOT NULL,

    Apellido VARCHAR(50) NOT NULL,

    Documento VARCHAR(20) NOT NULL UNIQUE,

    Telefono VARCHAR(20),

    Correo VARCHAR(100),

    Parentesco VARCHAR(30) NOT NULL
);
GO


/* ================================================================
   3. CREAR TABLA: PROFESORES
================================================================ */

CREATE TABLE Profesores
(
    IdProfesor INT IDENTITY(1,1) PRIMARY KEY,

    Nombre VARCHAR(50) NOT NULL,

    Apellido VARCHAR(50) NOT NULL,

    Documento VARCHAR(20) NOT NULL UNIQUE,

    Especialidad VARCHAR(100),

    Telefono VARCHAR(20),

    Correo VARCHAR(100),

    FechaIngreso DATE NOT NULL,

    Estado VARCHAR(20) DEFAULT 'ACTIVO'
);
GO


/* ================================================================
   4. CREAR TABLA: GRADOS
================================================================ */

CREATE TABLE Grados
(
    IdGrado INT IDENTITY(1,1) PRIMARY KEY,

    NombreGrado VARCHAR(50) NOT NULL,

    EdadMinima INT,

    EdadMaxima INT,

    Jornada VARCHAR(20)
);
GO


/* ================================================================
   5. CREAR TABLA: ESTUDIANTES
================================================================ */

CREATE TABLE Estudiantes
(
    IdEstudiante INT IDENTITY(1,1) PRIMARY KEY,

    Nombre VARCHAR(50) NOT NULL,

    Apellido VARCHAR(50) NOT NULL,

    Documento VARCHAR(20) UNIQUE,

    FechaNacimiento DATE NOT NULL,

    Genero VARCHAR(20),

    Direccion VARCHAR(150),

    Estado VARCHAR(20) DEFAULT 'ACTIVO',

    IdGrado INT NOT NULL,

    /* 
       LLAVE FORANEA

       Relaciona Estudiantes con Grados.
       Un estudiante pertenece a un grado.
    */

    CONSTRAINT FK_Estudiantes_Grados
        FOREIGN KEY (IdGrado)
        REFERENCES Grados(IdGrado)
);
GO


/* ================================================================
   6. RELACION ENTRE ESTUDIANTES Y PADRES
================================================================ */

/*
   Un estudiante puede tener varios padres o acudientes.

   Un padre tambien puede tener varios estudiantes.

   Por lo tanto tenemos una relacion:

             ESTUDIANTES
                   |
                   |
                   N
                   |
            EstudiantesPadres
                   |
                   N
                   |
                PADRES

   Es una relacion MUCHOS A MUCHOS.
*/

CREATE TABLE EstudiantesPadres
(
    IdEstudiante INT NOT NULL,

    IdPadre INT NOT NULL,

    EsAcudientePrincipal BIT DEFAULT 0,

    CONSTRAINT PK_EstudiantesPadres
        PRIMARY KEY (IdEstudiante, IdPadre),

    CONSTRAINT FK_EP_Estudiante
        FOREIGN KEY (IdEstudiante)
        REFERENCES Estudiantes(IdEstudiante),

    CONSTRAINT FK_EP_Padre
        FOREIGN KEY (IdPadre)
        REFERENCES PadresFamilia(IdPadre)
);
GO


/* ================================================================
   7. TABLA: ASIGNATURAS
================================================================ */

CREATE TABLE Asignaturas
(
    IdAsignatura INT IDENTITY(1,1) PRIMARY KEY,

    NombreAsignatura VARCHAR(100) NOT NULL,

    Descripcion VARCHAR(250),

    IntensidadHoraria INT,

    Estado VARCHAR(20) DEFAULT 'ACTIVA'
);
GO


/* ================================================================
   8. TABLA: PERIODOS ACADEMICOS
================================================================ */

CREATE TABLE Periodos
(
    IdPeriodo INT IDENTITY(1,1) PRIMARY KEY,

    NombrePeriodo VARCHAR(50) NOT NULL,

    FechaInicio DATE NOT NULL,

    FechaFin DATE NOT NULL
);
GO


/* ================================================================
   9. TABLA: PROFESORES - ASIGNATURAS
================================================================ */

/*
   Un profesor puede enseñar varias asignaturas.

   Una asignatura puede ser enseñada por varios profesores.

   Por eso necesitamos una tabla intermedia.
*/

CREATE TABLE ProfesoresAsignaturas
(
    IdProfesor INT NOT NULL,

    IdAsignatura INT NOT NULL,

    CONSTRAINT PK_ProfesoresAsignaturas
        PRIMARY KEY (IdProfesor, IdAsignatura),

    CONSTRAINT FK_PA_Profesor
        FOREIGN KEY (IdProfesor)
        REFERENCES Profesores(IdProfesor),

    CONSTRAINT FK_PA_Asignatura
        FOREIGN KEY (IdAsignatura)
        REFERENCES Asignaturas(IdAsignatura)
);
GO


/* ================================================================
   10. TABLA: CURSOS
================================================================ */

/*
   Un grado puede tener varios cursos.

   Ejemplo:

   Jardin
      |
      +---- Curso A
      +---- Curso B
*/

CREATE TABLE Cursos
(
    IdCurso INT IDENTITY(1,1) PRIMARY KEY,

    NombreCurso VARCHAR(50) NOT NULL,

    IdGrado INT NOT NULL,

    IdProfesorDirector INT,

    CONSTRAINT FK_Cursos_Grado
        FOREIGN KEY (IdGrado)
        REFERENCES Grados(IdGrado),

    CONSTRAINT FK_Cursos_Profesor
        FOREIGN KEY (IdProfesorDirector)
        REFERENCES Profesores(IdProfesor)
);
GO


/* ================================================================
   11. TABLA: MATRICULAS
================================================================ */

/*
   Esta tabla relaciona:

       ESTUDIANTE
           |
           |
       MATRICULA
           |
           |
         CURSO
*/

CREATE TABLE Matriculas
(
    IdMatricula INT IDENTITY(1,1) PRIMARY KEY,

    IdEstudiante INT NOT NULL,

    IdCurso INT NOT NULL,

    FechaMatricula DATE DEFAULT GETDATE(),

    Estado VARCHAR(20) DEFAULT 'ACTIVA',

    CONSTRAINT FK_Matricula_Estudiante
        FOREIGN KEY (IdEstudiante)
        REFERENCES Estudiantes(IdEstudiante),

    CONSTRAINT FK_Matricula_Curso
        FOREIGN KEY (IdCurso)
        REFERENCES Cursos(IdCurso)
);
GO


/* ================================================================
   12. TABLA: CALIFICACIONES
================================================================ */

CREATE TABLE Calificaciones
(
    IdCalificacion INT IDENTITY(1,1) PRIMARY KEY,

    IdEstudiante INT NOT NULL,

    IdAsignatura INT NOT NULL,

    IdPeriodo INT NOT NULL,

    Nota DECIMAL(3,2) NOT NULL,

    Observacion VARCHAR(250),

    CONSTRAINT FK_Calificacion_Estudiante
        FOREIGN KEY (IdEstudiante)
        REFERENCES Estudiantes(IdEstudiante),

    CONSTRAINT FK_Calificacion_Asignatura
        FOREIGN KEY (IdAsignatura)
        REFERENCES Asignaturas(IdAsignatura),

    CONSTRAINT FK_Calificacion_Periodo
        FOREIGN KEY (IdPeriodo)
        REFERENCES Periodos(IdPeriodo),

    /* La nota debe estar entre 1 y 5 */
    CONSTRAINT CK_Calificacion_Nota
        CHECK (Nota >= 1 AND Nota <= 5)
);
GO


/* ================================================================
   13. TABLA: ASISTENCIA
================================================================ */

CREATE TABLE Asistencia
(
    IdAsistencia INT IDENTITY(1,1) PRIMARY KEY,

    IdEstudiante INT NOT NULL,

    Fecha DATE NOT NULL,

    Estado VARCHAR(20) NOT NULL,

    Observacion VARCHAR(250),

    CONSTRAINT FK_Asistencia_Estudiante
        FOREIGN KEY (IdEstudiante)
        REFERENCES Estudiantes(IdEstudiante),

    CONSTRAINT CK_Asistencia_Estado
        CHECK (Estado IN ('PRESENTE','AUSENTE','EXCUSA'))
);
GO

/* ================================================================
   27. WHERE
================================================================ */

/*
   Mostrar estudiantes del grado 3.
*/

SELECT *
FROM Estudiantes
WHERE IdGrado = 3;
GO


/* ================================================================
   28. ORDER BY
================================================================ */

SELECT *
FROM Estudiantes
ORDER BY Apellido ASC;
GO


/* ================================================================
   29. INNER JOIN
================================================================ */

/*
   INNER JOIN muestra solamente los registros
   que tienen coincidencia en las dos tablas.

   En este ejemplo:

   ESTUDIANTES ---- GRADOS

   Solo aparecen estudiantes que tienen
   un grado relacionado.
*/

SELECT
    E.IdEstudiante,
    E.Nombre,
    E.Apellido,
    G.NombreGrado,
    G.Jornada
FROM Estudiantes E
INNER JOIN Grados G
    ON E.IdGrado = G.IdGrado;
GO


/* ================================================================
   30. INNER JOIN ENTRE VARIAS TABLAS
================================================================ */

/*
   Mostrar:

   Estudiante
   Curso
   Grado
   Profesor director
*/

SELECT
    E.Nombre AS Estudiante,
    E.Apellido,
    C.NombreCurso,
    G.NombreGrado,
    P.Nombre AS Profesor,
    P.Apellido AS ApellidoProfesor
FROM Estudiantes E
INNER JOIN Matriculas M
    ON E.IdEstudiante = M.IdEstudiante
INNER JOIN Cursos C
    ON M.IdCurso = C.IdCurso
INNER JOIN Grados G
    ON C.IdGrado = G.IdGrado
INNER JOIN Profesores P
    ON C.IdProfesorDirector = P.IdProfesor;
GO


/* ================================================================
   31. LEFT JOIN
================================================================ */

/*
   LEFT JOIN muestra TODOS los estudiantes,
   incluso si alguno no tiene una calificacion.

   Esto es muy importante en bases de datos.

   INNER JOIN:
   solamente coincidencias.

   LEFT JOIN:
   todos los registros de la tabla izquierda
   y las coincidencias de la derecha.
*/

SELECT
    E.Nombre,
    E.Apellido,
    C.Nota
FROM Estudiantes E
LEFT JOIN Calificaciones C
    ON E.IdEstudiante = C.IdEstudiante;
GO


/* ================================================================
   32. JOIN CON PADRES
================================================================ */

SELECT
    E.Nombre AS NombreEstudiante,
    E.Apellido AS ApellidoEstudiante,
    P.Nombre AS NombrePadre,
    P.Apellido AS ApellidoPadre,
    P.Telefono
FROM Estudiantes E
INNER JOIN EstudiantesPadres EP
    ON E.IdEstudiante = EP.IdEstudiante
INNER JOIN PadresFamilia P
    ON EP.IdPadre = P.IdPadre;
GO


/* ================================================================
   33. PROMEDIO DE CALIFICACIONES
================================================================ */

SELECT
    E.Nombre,
    E.Apellido,
    AVG(C.Nota) AS Promedio
FROM Estudiantes E
INNER JOIN Calificaciones C
    ON E.IdEstudiante = C.IdEstudiante
GROUP BY
    E.Nombre,
    E.Apellido;
GO


/* ================================================================
   34. GROUP BY
================================================================ */

/*
   Contar cuantos estudiantes existen
   por cada grado.
*/

SELECT
    G.NombreGrado,
    COUNT(E.IdEstudiante) AS CantidadEstudiantes
FROM Grados G
LEFT JOIN Estudiantes E
    ON G.IdGrado = E.IdGrado
GROUP BY
    G.NombreGrado;
GO

   37. DELETE

/* ================================================================
   40. VISTA: ESTUDIANTES Y GRADOS
================================================================ */

/*
   Una VISTA es una consulta almacenada.

   No guarda normalmente los datos duplicados.
   Guarda la consulta.

   Ventajas:

   - Facilita consultas frecuentes.
   - Mejora la organizacion.
   - Puede ocultar columnas sensibles.
   - Permite reutilizar consultas.

*/

CREATE VIEW vw_EstudiantesGrados
AS
SELECT
    E.IdEstudiante,
    E.Nombre,
    E.Apellido,
    G.NombreGrado,
    G.Jornada
FROM Estudiantes E
INNER JOIN Grados G
    ON E.IdGrado = G.IdGrado;
GO


/* Consultar la vista */

SELECT *
FROM vw_EstudiantesGrados;
GO


/* ================================================================
   41. VISTA DE CALIFICACIONES
================================================================ */

CREATE VIEW vw_NotasEstudiantes
AS
SELECT
    E.Nombre AS Estudiante,
    E.Apellido,
    A.NombreAsignatura,
    P.NombrePeriodo,
    C.Nota,
    C.Observacion
FROM Calificaciones C
INNER JOIN Estudiantes E
    ON C.IdEstudiante = E.IdEstudiante
INNER JOIN Asignaturas A
    ON C.IdAsignatura = A.IdAsignatura
INNER JOIN Periodos P
    ON C.IdPeriodo = P.IdPeriodo;
GO


SELECT *
FROM vw_NotasEstudiantes;
GO


/* ================================================================
   42. PROCEDIMIENTO ALMACENADO SIN PARAMETROS
================================================================ */

/*
   Un procedimiento almacenado permite guardar
   instrucciones SQL para reutilizarlas.

   Este procedimiento muestra todos los estudiantes.
*/

CREATE PROCEDURE sp_ListarEstudiantes
AS
BEGIN

    SELECT
        IdEstudiante,
        Nombre,
        Apellido,
        FechaNacimiento,
        Genero,
        Estado
    FROM Estudiantes
    ORDER BY Apellido;

END;
GO


/* Ejecutar procedimiento */

EXEC sp_ListarEstudiantes;
GO


/* ================================================================
   43. PROCEDIMIENTO ALMACENADO CON PARAMETRO
================================================================ */

/*
   Este procedimiento recibe el Id de un estudiante
   y muestra sus datos.
*/

CREATE PROCEDURE sp_BuscarEstudiante
    @IdEstudiante INT
AS
BEGIN

    SELECT
        E.IdEstudiante,
        E.Nombre,
        E.Apellido,
        G.NombreGrado
    FROM Estudiantes E
    INNER JOIN Grados G
        ON E.IdGrado = G.IdGrado
    WHERE E.IdEstudiante = @IdEstudiante;

END;
GO


/* Ejecutar */

EXEC sp_BuscarEstudiante @IdEstudiante = 5;
GO


/* ================================================================
   44. PROCEDIMIENTO PARA CONSULTAR NOTAS

CREATE PROCEDURE sp_NotasEstudiante
    @IdEstudiante INT
AS
BEGIN

    SELECT
        E.Nombre,
        E.Apellido,
        A.NombreAsignatura,
        P.NombrePeriodo,
        C.Nota,
        C.Observacion

    FROM Calificaciones C

    INNER JOIN Estudiantes E

    INNER JOIN Asignaturas A

/* ================================================================
   47. PROCEDIMIENTO PARA CONSULTAR ESTUDIANTES POR GRADO
================================================================ */

CREATE PROCEDURE sp_EstudiantesPorGrado

    @IdGrado INT

AS
BEGIN

    SELECT
        E.IdEstudiante,
        E.Nombre,
        E.Apellido,
        G.NombreGrado

    FROM Estudiantes E

    INNER JOIN Grados G
        ON E.IdGrado = G.IdGrado

    WHERE G.IdGrado = @IdGrado

    ORDER BY E.Apellido;

END;
GO

GO


/* ================================================================
   48. CONSULTA PARA ESTUDIANTES CON PROMEDIO MAYOR A 4
================================================================ */

SELECT
    E.Nombre,
    E.Apellido,
    AVG(C.Nota) AS Promedio
FROM Estudiantes E
INNER JOIN Calificaciones C
    ON E.IdEstudiante = C.IdEstudiante
GROUP BY
    E.Nombre,
    E.Apellido
HAVING AVG(C.Nota) >= 4;
GO


/* ================================================================
   49. CONSULTA DE PROFESORES Y ASIGNATURAS
================================================================ */

SELECT
    P.Nombre AS Profesor,
    P.Apellido,
    A.NombreAsignatura
FROM Profesores P
INNER JOIN ProfesoresAsignaturas PA
    ON P.IdProfesor = PA.IdProfesor
INNER JOIN Asignaturas A
    ON PA.IdAsignatura = A.IdAsignatura
ORDER BY
    P.Apellido;
GO


/* ================================================================
   50. LEFT JOIN PARA VER TODOS LOS PROFESORES
================================================================ */

/*
   Aqui queremos ver TODOS los profesores,
   incluso aquellos que actualmente no tengan
   una asignatura relacionada.
*/

SELECT
    P.Nombre,
    P.Apellido,
    A.NombreAsignatura
FROM Profesores P
LEFT JOIN ProfesoresAsignaturas PA
    ON P.IdProfesor = PA.IdProfesor
LEFT JOIN Asignaturas A
    ON PA.IdAsignatura = A.IdAsignatura;
GO


/* ================================================================
   51. CONSULTA DE ESTUDIANTES SIN CALIFICACIONES
================================================================ */

/*
   Muy buen ejemplo para explicar LEFT JOIN.

   Buscamos estudiantes que NO tienen notas.

   Cuando no existe coincidencia,
   C.IdCalificacion sera NULL.
*/

SELECT
    E.IdEstudiante,
    E.Nombre,
    E.Apellido
FROM Estudiantes E
LEFT JOIN Calificaciones C
    ON E.IdEstudiante = C.IdEstudiante
WHERE C.IdCalificacion IS NULL;
GO


/* ================================================================
   52. CONSULTA COMPLETA DEL SISTEMA ACADEMICO
================================================================ */

SELECT

    E.Nombre AS NombreEstudiante,
    E.Apellido AS ApellidoEstudiante,

    G.NombreGrado,

    C.NombreCurso,

    A.NombreAsignatura,

    P.NombrePeriodo,

    CAL.Nota,

    CAL.Observacion

FROM Estudiantes E

INNER JOIN Grados G
    ON E.IdGrado = G.IdGrado

INNER JOIN Matriculas M
    ON E.IdEstudiante = M.IdEstudiante

INNER JOIN Cursos C
    ON M.IdCurso = C.IdCurso

INNER JOIN Calificaciones CAL
    ON E.IdEstudiante = CAL.IdEstudiante

INNER JOIN Asignaturas A
    ON CAL.IdAsignatura = A.IdAsignatura

INNER JOIN Periodos P
    ON CAL.IdPeriodo = P.IdPeriodo

ORDER BY
    E.Apellido,
    E.Nombre;
GO
