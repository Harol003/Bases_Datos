/*====================================================================
CURSO PRACTICO DE MANTENIMIENTO Y RESPALDO DE BASES DE DATOS
SQL SERVER

EMPRESAS:
- POLIANIME
- MODATECH

OBJETIVOS

1. Consultar bases de datos.
2. Realizar Full Backup.
3. Realizar Differential Backup.
4. Realizar Transaction Log Backup.
5. Realizar Copy Only Backup.
6. Consultar informacion de los respaldos.
7. Consultar historial de respaldos.
8. Restaurar bases de datos.
9. Simular un desastre.
10. Diseñar un plan de mantenimiento.

NOTA

Antes de iniciar cree la carpeta:

C:\BackupsSQL

Todos los archivos de respaldo se almacenaran alli.

====================================================================*/


/*====================================================================
PASO 1
CONSULTAR LAS BASES DE DATOS EXISTENTES
====================================================================*/

SELECT *
FROM sys.databases;
GO


/*====================================================================
PASO 2
VERIFICAR QUE EXISTAN POLIANIMEDB Y MODATECHDB
====================================================================*/

SELECT name
FROM sys.databases
WHERE name IN ('PoliAnimeDB','ModaTechDB');
GO


/*====================================================================
PASO 3
FULL BACKUP DE POLIANIME

UN FULL BACKUP CREA UNA COPIA COMPLETA DE TODA LA BASE
DE DATOS.

INCLUYE:

- TABLAS
- DATOS
- VISTAS
- PROCEDIMIENTOS ALMACENADOS
- FUNCIONES
- USUARIOS
- RELACIONES

====================================================================*/

BACKUP DATABASE PoliAnimeDB

TO DISK='C:\BackupsSQL\PoliAnime_Full.bak'

WITH

FORMAT,

INIT,

NAME='Backup Completo PoliAnime',

DESCRIPTION='Respaldo completo de PoliAnimeDB',

STATS=10;
GO


/*====================================================================
PASO 4
FULL BACKUP DE MODATECH
====================================================================*/

BACKUP DATABASE ModaTechDB

TO DISK='C:\BackupsSQL\ModaTech_Full.bak'

WITH

FORMAT,

INIT,

NAME='Backup Completo ModaTech',

DESCRIPTION='Respaldo completo de ModaTechDB',

STATS=10;
GO


/*====================================================================
PASO 5
VERIFICAR EL TAMAÑO DE LAS BASES DE DATOS

DE LAS BASES EN EL SERVIDOR.
====================================================================*/

GO


/*====================================================================
PASO 17
CONSULTAR LOS ARCHIVOS CONTENIDOS EN EL BACKUP
====================================================================*/

FROM DISK='C:\BackupsSQL\PoliAnime_Full.bak';
GO


/*====================================================================
PASO 18
CONSULTAR HISTORIAL DE RESPALDOS

TYPE

D = FULL
I = DIFFERENTIAL
L = LOG
F = FILE

====================================================================*/

SELECT

database_name,
backup_start_date,
backup_finish_date,
type,
name

FROM msdb.dbo.backupset

ORDER BY backup_finish_date DESC;
GO


/*====================================================================
