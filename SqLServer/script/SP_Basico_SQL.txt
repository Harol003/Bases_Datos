/*
=========================================================
PASO 2 - CREAR TABLA CATEGORIAS

Una categoria permite clasificar los productos.

Ejemplos:
- Figuras
- Mangas
- Posters
- Llaveros

PRIMARY KEY:
Identifica de forma unica cada registro.

IDENTITY:
Genera automaticamente el consecutivo.
=========================================================
*/

CREATE TABLE Categorias
(
    id_categoria INT IDENTITY(1,1) PRIMARY KEY,

    nombre_categoria VARCHAR(100)
    NOT NULL
    UNIQUE
);
GO

/*
=========================================================
PASO 3 - CREAR TABLA PRODUCTOSANIME

Esta tabla almacena los productos vendidos
por PoliAnime.

FOREIGN KEY:
Relaciona cada producto con una categoria.
=========================================================
*/

CREATE TABLE ProductosAnime
(
    id_producto INT IDENTITY(1,1) PRIMARY KEY,

    nombre_producto VARCHAR(100)
    NOT NULL,

    precio DECIMAL(10,2)
    NOT NULL
    CHECK(precio > 0),

    stock INT
    DEFAULT 0
    CHECK(stock >= 0),

    id_categoria INT
    NOT NULL,

    CONSTRAINT FK_PRODUCTO_CATEGORIA
    FOREIGN KEY(id_categoria)
    REFERENCES Categorias(id_categoria)
);
GO

/*
=========================================================
PASO 4 - CREAR TABLA CLIENTES

UNIQUE:
No permite correos repetidos.
=========================================================
*/

CREATE TABLE Clientes
(
    id_cliente INT IDENTITY(1,1) PRIMARY KEY,

    nombre VARCHAR(100)
    NOT NULL,

    telefono VARCHAR(20),

    correo VARCHAR(100)
    UNIQUE,

    ciudad VARCHAR(100)
    DEFAULT 'Bogota'
);
GO

/*
=========================================================
PASO 5 - CREAR TABLA VENTAS

Relaciona clientes y productos.

Una venta pertenece a:
- Un cliente
- Un producto
=========================================================
*/

CREATE TABLE Ventas
(
    id_venta INT IDENTITY(1,1) PRIMARY KEY,

    fecha_venta DATE
    DEFAULT GETDATE(),

    cantidad INT
    CHECK(cantidad > 0),

    total DECIMAL(10,2)
    CHECK(total > 0),

    id_cliente INT,

    id_producto INT,

    CONSTRAINT FK_VENTA_CLIENTE
    FOREIGN KEY(id_cliente)
    REFERENCES Clientes(id_cliente),

    CONSTRAINT FK_VENTA_PRODUCTO
    FOREIGN KEY(id_producto)
    REFERENCES ProductosAnime(id_producto)
);
GO

/*
=========================================================
PASO 11 - INNER JOIN

Muestra solamente registros relacionados.
=========================================================
*/

SELECT
P.nombre_producto,
C.nombre_categoria
FROM ProductosAnime P
INNER JOIN Categorias C
ON P.id_categoria = C.id_categoria;
GO

/*
=========================================================
PASO 12 - INNER JOIN MULTIPLE

Mostrar ventas completas.
=========================================================
*/

SELECT
V.id_venta,
CL.nombre,
P.nombre_producto,
V.cantidad,
V.total
FROM Ventas V
INNER JOIN Clientes CL
ON V.id_cliente = CL.id_cliente
INNER JOIN ProductosAnime P
ON V.id_producto = P.id_producto;
GO

/*
=========================================================
PASO 13 - LEFT JOIN

Muestra todos los clientes aunque no tengan ventas.
=========================================================
*/

SELECT
C.nombre,
V.id_venta
FROM Clientes C
LEFT JOIN Ventas V
ON C.id_cliente = V.id_cliente;
GO

/*
=========================================================
PASO 14 - RIGHT JOIN

Muestra todas las ventas.
=========================================================
*/

SELECT
C.nombre,
V.id_venta
FROM Clientes C
RIGHT JOIN Ventas V
ON C.id_cliente = V.id_cliente;
GO

/*
=========================================================
PASO 15 - STORED PROCEDURE SIN PARAMETROS

Un Stored Procedure es un conjunto de
instrucciones almacenadas dentro de SQL Server.

Este procedimiento muestra todos los clientes.
=========================================================
*/

CREATE PROCEDURE sp_mostrar_clientes
AS
BEGIN

    SELECT *
    FROM Clientes;

END;
GO

EXEC sp_mostrar_clientes;
GO

/*
=========================================================
PASO 16 - STORED PROCEDURE CON PARAMETRO

@id recibe un valor enviado por el usuario.
=========================================================
*/

CREATE PROCEDURE sp_buscar_cliente
    @id INT
AS
BEGIN

    SELECT *
    FROM Clientes
    WHERE id_cliente = @id;

END;
GO

EXEC sp_buscar_cliente 1;
GO

/*
=========================================================
PASO 17 - STORED PROCEDURE CON VARIOS PARAMETROS
=========================================================
*/

CREATE PROCEDURE sp_buscar_producto
    @precio DECIMAL(10,2),
    @stock INT
AS
BEGIN

    SELECT *
    FROM ProductosAnime
    WHERE precio >= @precio
    AND stock >= @stock;

END;
GO

EXEC sp_buscar_producto 100000,5;
GO

/*
=========================================================
PASO 18 - STORED PROCEDURE CON VALOR POR DEFECTO
=========================================================
*/

CREATE PROCEDURE sp_stock_bajo
    @stock INT = 5
AS
BEGIN

    SELECT *
    FROM ProductosAnime
    WHERE stock <= @stock;

END;
GO

EXEC sp_stock_bajo;
GO

EXEC sp_stock_bajo 20;
GO

PASO 19 - CRUD MEDIANTE STORED PROCEDURES

CREATE
=========================================================
*/

CREATE PROCEDURE sp_insertar_cliente

    @nombre VARCHAR(100),
    @telefono VARCHAR(20),
    @correo VARCHAR(100),
    @ciudad VARCHAR(100)

AS
BEGIN

    INSERT INTO Clientes
    (
        nombre,
        telefono,
        correo,
        ciudad
    )
    VALUES
    (
        @nombre,
        @telefono,
        @correo,
        @ciudad
    );

END;
GO

EXEC sp_insertar_cliente
'Pedro Ramirez',
'3111111111',
'pedro@gmail.com',
'Bogota';
GO

CREATE PROCEDURE sp_actualizar_cliente

    @id INT,
    @telefono VARCHAR(20)

AS
BEGIN

    UPDATE Clientes

    SET telefono = @telefono

    WHERE id_cliente = @id;

END;
GO

'3999999999';
GO

CREATE PROCEDURE sp_eliminar_cliente

    @id INT

AS
BEGIN

    DELETE FROM Clientes
    WHERE id_cliente = @id;

END;
GO
