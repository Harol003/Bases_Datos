====================================================================*/

CREATE TABLE Clientes
(
    id_cliente INT IDENTITY(1,1) PRIMARY KEY,

    nombre VARCHAR(100)
    NOT NULL,

    telefono VARCHAR(20),

    correo VARCHAR(100)
    UNIQUE,

    ciudad VARCHAR(100)
);
GO


/*====================================================================
PASO 3
CREAR TABLA PRODUCTOS
====================================================================*/

CREATE TABLE Productos
(
    id_producto INT IDENTITY(1,1) PRIMARY KEY,

    nombre_producto VARCHAR(100)
    NOT NULL,

    precio DECIMAL(10,2)
    CHECK(precio > 0),

    stock INT
    CHECK(stock >= 0)
);
GO

CREATE PROCEDURE sp_buscar_cliente

    @id INT

AS
BEGIN

    SELECT *

    FROM Clientes

    WHERE id_cliente = @id;

END;
GO

/*====================================================================
SP 3
SP CON VARIOS PARAMETROS
====================================================================*/

CREATE PROCEDURE sp_buscar_producto

    @precio DECIMAL(10,2),
    @stock INT

AS
BEGIN

    SELECT *

    FROM Productos

    WHERE precio >= @precio
    AND stock >= @stock;

END;
GO


/*====================================================================
EJECUTAR SP CON VARIOS PARAMETROS
====================================================================*/

EXEC sp_buscar_producto
50000,
5;
GO


/*====================================================================
SP 4
PARAMETRO CON VALOR POR DEFECTO
====================================================================*/

CREATE PROCEDURE sp_stock_bajo

    @stock INT = 5

AS
BEGIN

    SELECT *

    FROM Productos

    WHERE stock <= @stock;

END;
GO

/*====================================================================

/*====================================================================
SP 5
PARAMETRO OUTPUT

DEVUELVE UN RESULTADO HACIA UNA VARIABLE
====================================================================*/

CREATE PROCEDURE sp_contar_clientes

    @cantidad INT OUTPUT

AS
BEGIN

    SELECT

    @cantidad = COUNT(*)

    FROM Clientes;

END;
GO


/*====================================================================
EJECUTAR PARAMETRO OUTPUT
====================================================================*/

DECLARE @total INT;

EXEC sp_contar_clientes
@total OUTPUT;

SELECT @total AS total_clientes;
GO

/*====================================================================
SP 9
MANEJO DE ERRORES CON TRY CATCH
====================================================================*/

CREATE PROCEDURE sp_dividir

    @numero1 INT,
    @numero2 INT

AS
BEGIN

    BEGIN TRY

        SELECT

        @numero1 / @numero2

        AS resultado;

    END TRY

    BEGIN CATCH

        SELECT

        ERROR_NUMBER() AS numero_error,

        ERROR_MESSAGE() AS mensaje_error;

    END CATCH

END;
GO


/*====================================================================
EJECUTAR SP CON ERROR
====================================================================*/

EXEC sp_dividir
10,
0;
GO


/*====================================================================
SP 10
TRANSACCIONES
====================================================================*/

CREATE TABLE CuentaA
(
    saldo DECIMAL(10,2)
);
GO

CREATE TABLE CuentaB
(
    saldo DECIMAL(10,2)
);
GO


INSERT INTO CuentaA VALUES(1000000);
INSERT INTO CuentaB VALUES(500000);
GO


CREATE PROCEDURE sp_transferencia

    @valor DECIMAL(10,2)

AS
BEGIN

    BEGIN TRY

        BEGIN TRAN;

        UPDATE CuentaA
        SET saldo = saldo - @valor;

        UPDATE CuentaB
        SET saldo = saldo + @valor;

        COMMIT TRAN;

    END TRY

    BEGIN CATCH

        ROLLBACK TRAN;

    END CATCH

END;
GO


/*====================================================================
EJECUTAR TRANSFERENCIA
====================================================================*/

EXEC sp_transferencia 100000;
GO


/*====================================================================
VERIFICAR SALDOS
====================================================================*/

SELECT * FROM CuentaA;
GO

SELECT * FROM CuentaB;
GO

CREATE PROCEDURE sp_sumar

    @numero1 INT,
    @numero2 INT

AS
BEGIN

    RETURN @numero1 + @numero2;

END;
GO

====================================================================*/

CREATE PROCEDURE sp_clientes_ciudad

    @ciudad VARCHAR(100) = NULL

AS
BEGIN

    SELECT *

    FROM Clientes

    WHERE ciudad = @ciudad
    OR @ciudad IS NULL;

END;
GO
